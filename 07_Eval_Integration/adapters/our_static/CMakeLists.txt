cmake_minimum_required(VERSION 3.16)
project(string_index_static_adapter LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
set(STATIC_IMPL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../02_Code_and_Experiments/validation_e25_20260918")
file(SHA256 "${STATIC_IMPL_DIR}/static_physical_index.hpp" CORE_DIGEST)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/our_static_adapter.cpp" CPP_DIGEST)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/our_static_adapter.h" HEADER_DIGEST)
string(SHA256 ADAPTER_DIGEST "${CPP_DIGEST}${HEADER_DIGEST}")
add_library(our_static_adapter STATIC our_static_adapter.cpp)
target_compile_features(our_static_adapter PUBLIC cxx_std_17)
target_include_directories(our_static_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${STATIC_IMPL_DIR}")
if(DEFINED STRING_INDEX_CONTRACT_INCLUDE_DIR)
  target_include_directories(our_static_adapter PUBLIC "${STRING_INDEX_CONTRACT_INCLUDE_DIR}")
else()
  target_include_directories(our_static_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../learned-index/src")
endif()
target_compile_options(our_static_adapter PUBLIC -march=x86-64-v3 -Wall -Wextra)
target_compile_definitions(our_static_adapter PRIVATE STATIC_CORE_SHA256="${CORE_DIGEST}" STATIC_ADAPTER_SHA256="${ADAPTER_DIGEST}")
option(STATIC_BUILD_ACCEPTANCE "Build shared interface acceptance" OFF)
if(STATIC_BUILD_ACCEPTANCE)
  add_executable(our_static_acceptance ../common/contract_acceptance.cpp)
  target_link_libraries(our_static_acceptance PRIVATE our_static_adapter)
  target_compile_definitions(our_static_acceptance PRIVATE ACCEPTANCE_FACTORY_HEADER="our_static_adapter.h" ACCEPTANCE_FACTORY_SYMBOL=make_our_static_index ACCEPTANCE_SAMPLE_PARAMETER=lookup)
endif()
