cmake_minimum_required(VERSION 3.19)
project(string_index_static_refined_adapter LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
get_filename_component(WORKSPACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DIAGNOSTICS "${WORKSPACE_ROOT}/07_Eval_Integration/diagnostics")
set(IMPL "${WORKSPACE_ROOT}/02_Code_and_Experiments/validation_e25_20260918")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/benchmark-adapter.json" MANIFEST)
set(CORE_SOURCES "${DIAGNOSTICS}/collision_refined_index.hpp" "${DIAGNOSTICS}/collision_refinement.hpp"
    "${IMPL}/physical_locator_index.hpp" "${IMPL}/static_physical_index.hpp")
set(BUNDLE_HEX "")
set(I 0)
foreach(SOURCE IN LISTS CORE_SOURCES)
  file(SHA256 "${SOURCE}" ACTUAL)
  file(RELATIVE_PATH RELATIVE "${WORKSPACE_ROOT}" "${SOURCE}")
  string(JSON DECLARED GET "${MANIFEST}" source_identity core_sources ${I} sha256)
  string(JSON DECLARED_PATH GET "${MANIFEST}" source_identity core_sources ${I} path)
  if(NOT ACTUAL STREQUAL DECLARED OR NOT RELATIVE STREQUAL DECLARED_PATH)
    message(FATAL_ERROR "Manifest implementation identity is stale: ${SOURCE}")
  endif()
  string(APPEND BUNDLE_HEX "${ACTUAL}")
  math(EXPR I "${I}+1")
endforeach()
string(SHA256 BUNDLE "${BUNDLE_HEX}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/our_static_refined_adapter.cpp" CPP)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/our_static_refined_adapter.h" HEADER)
string(SHA256 ADAPTER "${CPP}${HEADER}")
string(JSON DECLARED_BUNDLE GET "${MANIFEST}" source_identity core_bundle_sha256)
string(JSON DECLARED_ADAPTER GET "${MANIFEST}" source_identity adapter_digest)
if(NOT BUNDLE STREQUAL DECLARED_BUNDLE OR NOT ADAPTER STREQUAL DECLARED_ADAPTER)
  message(FATAL_ERROR "Manifest bundle or adapter identity is stale")
endif()
add_library(our_static_refined_adapter STATIC our_static_refined_adapter.cpp)
target_compile_features(our_static_refined_adapter PUBLIC cxx_std_17)
target_include_directories(our_static_refined_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${DIAGNOSTICS}" "${IMPL}")
if(DEFINED STRING_INDEX_CONTRACT_INCLUDE_DIR)
  target_include_directories(our_static_refined_adapter PUBLIC "${STRING_INDEX_CONTRACT_INCLUDE_DIR}")
else()
  target_include_directories(our_static_refined_adapter PUBLIC "${WORKSPACE_ROOT}/07_Eval_Integration/learned-index/src")
endif()
target_compile_options(our_static_refined_adapter PUBLIC -march=x86-64-v3)
target_compile_definitions(our_static_refined_adapter PRIVATE REFINED_CORE_BUNDLE_SHA256="${BUNDLE}" REFINED_ADAPTER_SHA256="${ADAPTER}")
option(REFINED_BUILD_ACCEPTANCE "Build shared v1 acceptance" OFF)
if(REFINED_BUILD_ACCEPTANCE)
  add_executable(our_static_refined_acceptance ../common/contract_acceptance.cpp)
  target_link_libraries(our_static_refined_acceptance PRIVATE our_static_refined_adapter)
  target_compile_definitions(our_static_refined_acceptance PRIVATE ACCEPTANCE_FACTORY_HEADER="our_static_refined_adapter.h"
    ACCEPTANCE_FACTORY_SYMBOL=make_our_static_refined_index ACCEPTANCE_SAMPLE_PARAMETER=lookup)
endif()
